class Solution {
public:
    string evaluate(string s, vector<vector<string>>& k) {
        unordered_map<string, string>mp;
        int n = s.size();
        int m = k.size();

        for(int i=0;i<m;i++){
            mp[k[i][0]] = k[i][1];
        }
        string ans = "";
        int i = 0;
        while(i<n){
            if(s[i]!='(') ans.push_back(s[i]);
            else{
                i++;
                string key = "";
                while(s[i]!=')'){
                    key.push_back(s[i++]);
                }
                if(mp.find(key)!=mp.end()){
                    ans+=mp[key];
                }
                else ans+='?';
            }
            i++;
